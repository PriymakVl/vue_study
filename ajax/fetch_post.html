Отправка данных на сервер (POST-запрос) из дочернего компонента

  1. Для отправки данных методом POST вторым аргументом в fetch(url, options) передается объект настроек.
  2. В настройках обязательно указываются method: 'POST', заголовок 'Content-Type': 'application/json' и тело запроса, преобразованное в строку через JSON.stringify().
  3. После успешного ответа сервера дочерний компонент уведомляет родителя через defineEmits, чтобы добавить созданный элемент в общий список без перезагрузки страницы. 
<!-- src/components/PostForm.vue -->
<template>
  <form @submit.prevent="createPost" class="post-form">
    <input v-model.trim="title" placeholder="Заголовок записи" required />
    <button type="submit" :disabled="isSending">
      {{ isSending ? 'Отправка...' : 'Добавить запись' }}
    </button>
    <p v-if="error" class="error">{{ error }}</p>
  </form>
</template>

<script setup>
import { ref } from 'vue';

const emit = defineEmits(['created']);

const title = ref('');
const isSending = ref(false);
const error = ref(null);

const createPost = async () => {
  isSending.value = true;
  error.value = null;

  try {
    const response = await fetch('https://jsonplaceholder.typicode.com/posts', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json; charset=UTF-8'
      },
      body: JSON.stringify({
        title: title.value,
        userId: 1
      })
    });

    if (!response.ok) {
      throw new Error(`Не удалось сохранить: ${response.status}`);
    }

    const createdItem = await response.json();
    emit('created', createdItem);
    title.value = '';
  } catch (err) {
    error.value = err.message;
  } finally {
    isSending.value = false;
  }
};
</script>
<!-- src/App.vue -->
<template>
  <div class="container">
    <PostForm @created="addPost" />

    <ul>
      <li v-for="post in posts" :key="post.id">
        {{ post.title }}
      </li>
    </ul>
  </div>
</template>

<script setup>
import { ref } from 'vue';
import PostForm from './components/PostForm.vue';

const posts = ref([]);

const addPost = (newPost) => {
  posts.value.unshift(newPost);
};
</script>
  1. headers и JSON.stringify() — сервер должен знать, что ему передают JSON-строку, иначе тело запроса может не распарситься.
  2. emit('created', createdItem) — передает полученный от сервера объект (уже с присвоенным id) в родительский компонент по событию @created="addPost".
