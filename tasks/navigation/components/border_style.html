Border Style — Vue Task

Задание

  1. При выборе соответствующей радиокнопки динамически изменять стиль рамки блока.
Демо

Решение задачи

App.vue

Главный компонент управления состоянием стиля границы

<template>
<div id="app">
  <BorderControls v-model="borderStyle" />
  <BorderBox :style-type="borderStyle" />
</div>
</template>

<script>
import BorderControls from './components/BorderControls.vue';
import BorderBox from './components/BorderBox.vue';

export default {
name: 'App',
components: {
  BorderControls,
  BorderBox
},
data() {
  return {
    borderStyle: 'solid'
  };
}
};
</script>

<style>
#app {
display: flex;
flex-direction: column;
justify-content: center;
align-items: center;
height: 100vh;
overflow: hidden;
}
</style>

components/BorderControls.vue

Компонент с набором радио-кнопок для выбора стиля рамки


  <template>
    <div class="controls">
      <RadioLabel value="solid" :model-value="modelValue" @update:model-value="updateValue">
        Solid
      </RadioLabel>
  
      <RadioLabel value="dashed" :model-value="modelValue" @update:model-value="updateValue">
        Dashed
      </RadioLabel>
  
      <RadioLabel value="dotted" :model-value="modelValue" @update:model-value="updateValue">
        Dotted
      </RadioLabel>
    </div>
  </template>
  
  <script>
  import RadioLabel from './RadioLabel.vue';
  
  export default {
    name: 'BorderControls',
    components: {
      RadioLabel
    },
    props: {
      modelValue: {
        type: String,
        required: true
      }
    },
    emits: ['update:modelValue'],
    methods: {
      updateValue(val) {
        this.$emit('update:modelValue', val);
      }
    }
  };
  </script>

components/BorderBox.vue

Компонент отображения блока с динамическим стилем границы

<template>
    <div class="box" :style="{ borderStyle: styleType }"></div>
  </template>
  
  <script>
  export default {
    name: 'BorderBox',
    props: {
      styleType: {
        type: String,
        default: 'solid'
      }
    }
  };
  </script>
  
  <style scoped>
  .box {
    width: 200px;
    height: 100px;
    border: 4px solid black;
    margin-top: 20px;
  }
  </style>

components/RadioLabel.vue

Переиспользуемый компонент кастомной радио-кнопки


    <template>
      <label>
        <input 
          type="radio" 
          :value="value" 
          :checked="modelValue === value"
          @change="$emit('update:modelValue', value)" 
        />
        <slot></slot>
      </label>
    </template>
    
    <script>
    export default {
      name: 'RadioLabel',
      props: {
        modelValue: {
          type: String,
          required: true
        },
        value: {
          type: String,
          required: true
        }
      },
      emits: ['update:modelValue']
    };
    </script>
    
    <style scoped>
    label {
      margin-right: 10px;
      cursor: pointer;
    }
    </style>
