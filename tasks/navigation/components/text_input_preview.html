components/TextInput.vue

Компонент текстового поля ввода с эмитом событий

<template>
  <input 
    type="text" 
    :value="modelValue"
    @input="$emit('update:modelValue', $event.target.value)"
    placeholder="Введите текст"
  />
</template>

<script>
export default {
  name: 'TextInput',
  props: {
    modelValue: {
      type: String,
      default: ''
    }
  },
  emits: ['update:modelValue']
};
</script>
