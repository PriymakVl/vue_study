Todo List — Vue Task

Задание

  1. Написать скрипт для составления списка задач
  2. При клике по добавленной задаче она должна быть удалена из списка
Демо

Решение задачи (Компоненты Vue)

App.vue

Главный компонент управления состоянием списка задач

<template>
  <div class="container">
    <h1>Список задач</h1>
    
    <TaskInput @add-task="addTask" />

    <ul>
      <TaskItem 
        v-for="(task, index) in tasks" 
        :key="index" 
        @remove="deleteTask(index)"
      >
        {{ task }}
      </TaskItem>
    </ul>
  </div>
</template>

<script>
import TaskInput from './components/TaskInput.vue';
import TaskItem from './components/TaskItem.vue';

export default {
  name: 'App',
  components: {
    TaskInput,
    TaskItem
  },
  data() {
    return {
      tasks: []
    };
  },
  methods: {
    addTask(taskText) {
      this.tasks.push(taskText);
    },
    deleteTask(index) {
      this.tasks.splice(index, 1);
    }
  }
};
</script>

components/TaskInput.vue

Компонент ввода новой задачи и отправки события добавления

<template>
  <div class="task-input">
    <input 
      type="text" 
      v-model="newTask" 
      placeholder="Введите задачу"
      @keyup.enter="submitTask"
    >
    <button @click="submitTask">Добавить</button>
  </div>
</template>

<script>
export default {
  name: 'TaskInput',
  emits: ['add-task'],
  data() {
    return {
      newTask: ''
    };
  },
  methods: {
    submitTask() {
      if (this.newTask.trim() !== '') {
        this.$emit('add-task', this.newTask.trim());
        this.newTask = '';
      }
    }
  }
};
</script>
