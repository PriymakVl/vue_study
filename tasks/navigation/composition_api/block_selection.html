Block Selection — Vue Task

Задание

  1. Создать интерактивный список блоков с выбором номера через `select`.
  2. При выборе пункта менять значение соответствующего блока на его порядковый номер или сбрасывать все блоки к исходному состоянию.
Демо

Решение задачи (Компоненты Vue)

App.vue

Главный компонент управления состоянием блоков и выбора

<template>
  <main>
    <NumberSelect
      v-model="selected"
      :total="blocks.length"
      @update-block="handleUpdateBlock"
    />

    <div class="grid">
      <BlockSquare
        v-for="(block, index) in blocks"
        :key="index"
        :content="block"
      />
    </div>
  </main>
</template>

<script setup>
import { ref } from 'vue';
import NumberSelect from './components/NumberSelect.vue';
import BlockSquare from './components/BlockSquare.vue';

const selected = ref('');
const blocks = ref(Array(5).fill('блок'));

function handleUpdateBlock(val) {
  if (val === 'reset') {
    blocks.value = blocks.value.map(() => 'блок');
    return;
  }

  const index = Number(val);
  if (!isNaN(index)) {
    blocks.value.splice(index, 1, index + 1);
  }
}
</script>

components/NumberSelect.vue

Компонент выбора номера блока или сброса значения

<template>
  <select :value="modelValue" @change="onChange">
    <option disabled value="">Выберите номер</option>
    <option
      v-for="(_, index) in total"
      :key="index"
      :value="index"
    >
      {{ index + 1 }}
    </option>
    <option value="reset">Сбросить</option>
  </select>
</template>

<script setup>
defineProps({
  modelValue: {
    type: [String, Number],
    default: ''
  },
  total: {
    type: Number,
    required: true
  }
});

const emit = defineEmits(['update:modelValue', 'update-block']);

function onChange(event) {
  const val = event.target.value;
  emit('update:modelValue', val);
  emit('update-block', val);
}
</script>
