Крестики-нолики — Vue Task

Задание

  1. Реализовать игру «Крестики-нолики» на игровом поле 3×3 против компьютера (игрок ставит «X», компьютер отвечает случайным ходом «O» с небольшой задержкой).
  2. Определять и подсвечивать выигрышную комбинацию из трёх символов в ряд, колонку или по диагонали, а также обрабатывать ничью.
  3. Добавить кнопку «Новая игра» для сброса текущего состояния игрового поля и перезапуска партии.

Решение задачи

App.vue

Главный компонент управления состоянием игры, ходами компьютера и проверкой победы

<template>
  <div id="app">
    <h1>Крестики-нолики</h1>
    
    <GameBoard 
      :board="board" 
      :winner-combo="winnerCombo" 
      @move="makeMove" 
    />

    <h2 v-if="winner">🏆 Победитель: {{ winner }}</h2>
    <h2 v-else-if="isDraw">🤝 Ничья!</h2>
    
    <button @click="resetGame">🔄 Новая игра</button>
  </div>
</template>

<script setup>
import { ref, computed } from 'vue';
import GameBoard from './components/Board.vue';

const board = ref(Array(9).fill(""));
const currentPlayer = ref("X");
const winner = ref(null);
const winnerCombo = ref([]);

const isDraw = computed(() => {
  return board.value.every(cell => cell !== "") && !winner.value;
});

function checkWinner() {
  const combos = [
    [0, 1, 2], [3, 4, 5], [6, 7, 8], // ряды
    [0, 3, 6], [1, 4, 7], [2, 5, 8], // колонки
    [0, 4, 8], [2, 4, 6]             // диагонали
  ];
  for (const combo of combos) {
    const [a, b, c] = combo;
    if (
      board.value[a] &&
      board.value[a] === board.value[b] &&
      board.value[a] === board.value[c]
    ) {
      winnerCombo.value = combo;
      return true;
    }
  }
  return false;
}

function computerMove() {
  if (winner.value) return;

  const emptyCells = board.value
    .map((cell, i) => (cell === "" ? i : null))
    .filter(i => i !== null);

  if (emptyCells.length === 0) return;

  const randomIndex = emptyCells[Math.floor(Math.random() * emptyCells.length)];
  board.value[randomIndex] = "O";

  if (checkWinner()) {
    winner.value = "O";
  } else {
    currentPlayer.value = "X";
  }
}

function makeMove(index) {
  if (board.value[index] || winner.value || currentPlayer.value !== "X") return;
  board.value[index] = "X";

  if (checkWinner()) {
    winner.value = "X";
    return;
  }

  currentPlayer.value = "O";
  setTimeout(computerMove, 500);
}

function resetGame() {
  board.value = Array(9).fill("");
  currentPlayer.value = "X";
  winner.value = null;
  winnerCombo.value = [];
}
</script>

components/Board.vue

Компонент игрового поля 3×3, отрисовывающий сетку ячеек и передающий события хода

<template>
  <div class="board">
    <GameCell
      v-for="(cell, index) in board"
      :key="index"
      :value="cell"
      :is-winner="winnerCombo.includes(index)"
      @cell-click="emit('move', index)"
    />
  </div>
</template>

<script setup>
import GameCell from './Cell.vue';

defineProps({
  board: {
    type: Array,
    required: true
  },
  winnerCombo: {
    type: Array,
    default: () => []
  }
});

const emit = defineEmits(['move']);
</script>
